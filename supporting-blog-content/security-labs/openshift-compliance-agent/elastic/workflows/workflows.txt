    type: workflow.output

{
  "id": "openshift-compliance-trigger-scan",
  "yaml": """
name: openshift-compliance-trigger-scan
description: Trigger a re-run of one ComplianceScan on a live OpenShift cluster by setting the rescan annotation.
enabled: true
tags: ["openshift-compliance"]

consts:
  namespace: "openshift-compliance"
  api_version: "compliance.openshift.io/v1alpha1"

triggers:
  - type: manual
    inputs:
      - name: cluster
        type: string
        required: true
        description: "Cluster name as stored in Elasticsearch, matching the Kubernetes connector name."
      - name: scan_name
        type: string
        required: true
        description: "Exact ComplianceScan name, for example ocp4-high."

outputs:
  - name: cluster
    type: string
  - name: scan_name
    type: string
  - name: previous_phase
    type: string
  - name: previous_result
    type: string
  - name: rescan_requested
    type: boolean

settings:
  timeout: "2m"

steps:
  - name: get_scan
    type: kubernetes.getResource
    connector-id: "{{ inputs.cluster }}"
    with:
      apiVersion: "{{ consts.api_version }}"
      resource: "compliancescans"
      name: "{{ inputs.scan_name }}"
      namespace: "{{ consts.namespace }}"
  - name: guard_phase
    type: if
    condition: "not steps.get_scan.output.status.phase : DONE"
    steps:
      - name: scan_not_done
        type: workflow.fail
        with:
          message: "Scan '{{ inputs.scan_name }}' on cluster '{{ inputs.cluster }}' is not in the DONE phase. Not re-running it."
          reason: "scan_not_done"
  - name: annotate_rescan
    type: kubernetes.patchResource
    connector-id: "{{ inputs.cluster }}"
    with:
      apiVersion: "{{ consts.api_version }}"
      resource: "compliancescans"
      name: "{{ inputs.scan_name }}"
      namespace: "{{ consts.namespace }}"
      patchType: "merge"
      patch:
        metadata:
          annotations:
            compliance.openshift.io/rescan: ""
  - name: return_output
    type: workflow.output
    with:
      cluster: "{{ inputs.cluster }}"
      scan_name: "{{ inputs.scan_name }}"
      previous_phase: "{{ steps.get_scan.output.status.phase }}"
      previous_result: "{{ steps.get_scan.output.status.result }}"
      rescan_requested: true
"""
}

name: openshift-compliance-set-remediation-state
description: Apply or revert one ComplianceRemediation on a live OpenShift cluster. Dry run first.
enabled: true
tags: ["openshift-compliance"]

consts:
  namespace: "openshift-compliance"
  api_version: "compliance.openshift.io/v1alpha1"

triggers:
  - type: manual
    inputs:
      - name: cluster
        type: string
        required: true
        description: "Cluster name as stored in Elasticsearch, matching the Kubernetes connector name."
      - name: remediation_name
        type: string
        required: true
        description: "Exact ComplianceRemediation name, for example ocp4-cis-audit-profile-set."
      - name: apply
        type: boolean
        required: true
        description: "true to apply the remediation, false to revert it."

outputs:
  - name: cluster
    type: string
  - name: remediation_name
    type: string
  - name: apply_requested
    type: string
  - name: before
    type: string
  - name: after
    type: string

settings:
  timeout: "2m"

steps:
  - name: get_before
    type: kubernetes.getResource
    connector-id: "{{ inputs.cluster }}"
    with:
      apiVersion: "{{ consts.api_version }}"
      resource: "complianceremediations"
      name: "{{ inputs.remediation_name }}"
      namespace: "{{ consts.namespace }}"
  - name: dry_run_patch
    type: kubernetes.patchResource
    connector-id: "{{ inputs.cluster }}"
    with:
      apiVersion: "{{ consts.api_version }}"
      resource: "complianceremediations"
      name: "{{ inputs.remediation_name }}"
      namespace: "{{ consts.namespace }}"
      patchType: "merge"
      dryRun: true
      patch:
        spec:
          apply: "${{ inputs.apply }}"
  - name: patch_resource
    type: kubernetes.patchResource
    connector-id: "{{ inputs.cluster }}"
    with:
      apiVersion: "{{ consts.api_version }}"
      resource: "complianceremediations"
      name: "{{ inputs.remediation_name }}"
      namespace: "{{ consts.namespace }}"
      patchType: "merge"
      patch:
        spec:
          apply: "${{ inputs.apply }}"
  - name: get_after
    type: kubernetes.getResource
    connector-id: "{{ inputs.cluster }}"
    with:
      apiVersion: "{{ consts.api_version }}"
      resource: "complianceremediations"
      name: "{{ inputs.remediation_name }}"
      namespace: "{{ consts.namespace }}"
  - name: return_output
    type: workflow.output
    with:
      cluster: "{{ inputs.cluster }}"
      remediation_name: "{{ inputs.remediation_name }}"
      apply_requested: "{{ inputs.apply }}"
      before: "{{ steps.get_before.output | json }}"
      after: "{{ steps.get_after.output | json }}"
"""
